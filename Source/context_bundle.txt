106:        // Returns an 8-bit mask where bit `lane` is 1 if visible, 0 if culled
107:        [[nodiscard]] inline uint32_t Cull8AABB(
108:            __m256 cX, __m256 cY, __m256 cZ,
109:            __m256 eX, __m256 eY, __m256 eZ) const noexcept
110:        {
111:            __m256 anyPlaneCulled{ _mm256_setzero_ps() };
113:            for (std::size_t i{ 0 }; i < planeCount; ++i)
114:            {
115:                const __m256 dot{ _mm256_add_ps(

119:                const __m256 rEff{ _mm256_add_ps(
120:                    _mm256_mul_ps(planes[i].absX, eX),
121:                    _mm256_add_ps(_mm256_mul_ps(planes[i].absY, eY), _mm256_mul_ps(planes[i].absZ, eZ))) };

256:    [[nodiscard]] uint16_t HashMaterial(const Material* mat) noexcept
257:    {
258:        if (!mat) return 0;
260:        // 16-byte aligned payload capturing surface parameters
261:        struct alignas(16) MaterialKeyData
262:        {
263:            void* baseMap{ nullptr };
264:            void* normalMap{ nullptr };
265:            void* emissiveMap{ nullptr };
266:            void* occlusionMap{ nullptr };
267:            void* metalnessRoughnessMap{ nullptr };
268:            DirectX::XMFLOAT3 emissiveColor{ 0.0f, 0.0f, 0.0f };
269:            float roughness{ 0.5f };
270:            float metalness{ 0.0f };
271:            float occlusionStrength{ 1.0f };
272:            float alphaCutoff{ 0.5f };
273:            int alphaMode{ 0 };
274:        };
276:        // Value-initialization guarantees all internal padding bytes are zeroed
277:        MaterialKeyData data{};
279:        // Common to all forward surface shaders (baseColor streamed per-instance in Slot 1)
280:        data.baseMap = mat->baseMap.Get();
281:        data.alphaCutoff = mat->alphaCutoff;
282:        data.alphaMode = static_cast<int>(mat->alphaMode);
284:        // Scope hashing strictly to properties evaluated by the active shader permutation
285:        switch (static_cast<ShaderId>(mat->shaderId))
286:        {
287:        case ShaderId::Basic:
288:            // Basic shader ignores lighting, normal mapping, PBR maps, and outlines
289:            break;
291:        case ShaderId::Lambert:
292:            data.emissiveColor = mat->emissiveColor;
293:            break;
295:        case ShaderId::Phong:
296:            data.normalMap = mat->normalMap.Get();
297:            data.emissiveColor = mat->emissiveColor;
298:            data.roughness = mat->roughness;
299:            break;
301:        case ShaderId::Pbr:
302:            data.normalMap = mat->normalMap.Get();
303:            data.emissiveMap = mat->emissiveMap.Get();
304:            data.occlusionMap = mat->occlusionMap.Get();
305:            data.metalnessRoughnessMap = mat->metalnessRoughnessMap.Get();
306:            data.emissiveColor = mat->emissiveColor;
307:            data.roughness = mat->roughness;
308:            data.metalness = mat->metalness;
309:            data.occlusionStrength = mat->occlusionStrength;
310:            break;
312:        case ShaderId::Toon:
313:            // Surface pass only evaluates diffuse, emissive, and stepped specular
314:            data.emissiveColor = mat->emissiveColor;
315:            data.roughness = mat->roughness;
316:            break;
318:        default:
319:            break;
320:        }
322:        const std::uint64_t hash{ Core::BufferHash(&data, sizeof(MaterialKeyData)) };
323:        return static_cast<uint16_t>(hash ^ (hash >> 16) ^ (hash >> 32) ^ (hash >> 48));
324:    }

466:void ModelRenderer::SoftwareOcclusionBuffer::RasterizeTriangle(
467:    float x0, float y0, float z0,
468:    float x1, float y1, float z1,
469:    float x2, float y2, float z2) noexcept
470:{
471:    float area{ (x1 - x0) * (y2 - y0) - (y1 - y0) * (x2 - x0) };
472:    if (std::abs(area) < 0.0001f)
473:    {
474:        return;
475:    }
477:    // Two-sided support for walls: ensure consistent clockwise rasterization winding
478:    if (area < 0.0f)
479:    {
480:        std::swap(x1, x2);
481:        std::swap(y1, y2);
482:        std::swap(z1, z2);
483:        area = -area;
484:    }
486:    int minX{ static_cast<int>(std::floor((std::min)({ x0, x1, x2 }))) };
487:    int maxX{ static_cast<int>(std::ceil((std::max)({ x0, x1, x2 }))) };
488:    int minY{ static_cast<int>(std::floor((std::min)({ y0, y1, y2 }))) };
489:    int maxY{ static_cast<int>(std::ceil((std::max)({ y0, y1, y2 }))) };
491:    minX = std::clamp(minX, 0, kWidth - 1);
492:    maxX = std::clamp(maxX, 0, kWidth - 1);
493:    minY = std::clamp(minY, 0, kHeight - 1);
494:    maxY = std::clamp(maxY, 0, kHeight - 1);
496:    if (minX > maxX || minY > maxY) return;
498:    const float invArea{ 1.0f / area };
500:    for (int y{ minY }; y <= maxY; ++y)
501:    {
502:        const float py{ static_cast<float>(y) + 0.5f };
503:        const int rowOffset{ y * kWidth };
505:        for (int x{ minX }; x <= maxX; ++x)
506:        {
507:            const float px{ static_cast<float>(x) + 0.5f };
508:            const float w0{ ((x1 - px) * (y2 - py) - (y1 - py) * (x2 - px)) * invArea };
509:            const float w1{ ((x2 - px) * (y0 - py) - (y2 - py) * (x0 - px)) * invArea };
510:            const float w2{ 1.0f - w0 - w1 };
512:            if (w0 >= 0.0f && w1 >= 0.0f && w2 >= 0.0f)
513:            {
514:                const float depth{ w0 * z0 + w1 * z1 + w2 * z2 };
515:                const int idx{ rowOffset + x };
516:                if (depth < depthBuffer[idx])
517:                {
518:                    depthBuffer[idx] = depth;
519:                }
520:            }
521:        }
522:    }

525:void ModelRenderer::SoftwareOcclusionBuffer::ClipAndRasterizeTriangle(
526:    const ClipVertex& v0,
527:    const ClipVertex& v1,
528:    const ClipVertex& v2) noexcept
529:{
530:    constexpr float kNearW{ 0.05f };
532:    auto projectAndDraw = [this](const ClipVertex& a, const ClipVertex& b, const ClipVertex& c) noexcept
533:        {
534:            const float invWa{ 1.0f / a.w };
535:            const float invWb{ 1.0f / b.w };
536:            const float invWc{ 1.0f / c.w };
538:            const float sx0{ (a.x * invWa * 0.5f + 0.5f) * static_cast<float>(kWidth - 1) };
539:            const float sy0{ (-a.y * invWa * 0.5f + 0.5f) * static_cast<float>(kHeight - 1) };
540:            const float sz0{ a.z * invWa };
542:            const float sx1{ (b.x * invWb * 0.5f + 0.5f) * static_cast<float>(kWidth - 1) };
543:            const float sy1{ (-b.y * invWb * 0.5f + 0.5f) * static_cast<float>(kHeight - 1) };
544:            const float sz1{ b.z * invWb };
546:            const float sx2{ (c.x * invWc * 0.5f + 0.5f) * static_cast<float>(kWidth - 1) };
547:            const float sy2{ (-c.y * invWc * 0.5f + 0.5f) * static_cast<float>(kHeight - 1) };
548:            const float sz2{ c.z * invWc };
550:            RasterizeTriangle(sx0, sy0, sz0, sx1, sy1, sz1, sx2, sy2, sz2);
551:        };
553:    auto lerpVertex = [](const ClipVertex& a, const ClipVertex& b, float t) noexcept -> ClipVertex
554:        {
555:            return {
556:                a.x + t * (b.x - a.x),
557:                a.y + t * (b.y - a.y),
558:                a.z + t * (b.z - a.z),
559:                a.w + t * (b.w - a.w)
560:            };
561:        };
563:    const bool in0{ v0.w >= kNearW };
564:    const bool in1{ v1.w >= kNearW };
565:    const bool in2{ v2.w >= kNearW };
566:    const int inCount{ (in0 ? 1 : 0) + (in1 ? 1 : 0) + (in2 ? 1 : 0) };
568:    if (inCount == 3)
569:    {
570:        projectAndDraw(v0, v1, v2);
571:    }
572:    else if (inCount == 1)
573:    {
574:        if (in0)
575:        {
576:            const float t1{ (kNearW - v0.w) / (v1.w - v0.w) };
577:            const float t2{ (kNearW - v0.w) / (v2.w - v0.w) };
578:            projectAndDraw(v0, lerpVertex(v0, v1, t1), lerpVertex(v0, v2, t2));
579:        }
580:        else if (in1)
581:        {
582:            const float t0{ (kNearW - v1.w) / (v0.w - v1.w) };
583:            const float t2{ (kNearW - v1.w) / (v2.w - v1.w) };
584:            projectAndDraw(lerpVertex(v1, v0, t0), v1, lerpVertex(v1, v2, t2));
585:        }
586:        else
587:        {
588:            const float t0{ (kNearW - v2.w) / (v0.w - v2.w) };
589:            const float t1{ (kNearW - v2.w) / (v1.w - v2.w) };
590:            projectAndDraw(lerpVertex(v2, v0, t0), lerpVertex(v2, v1, t1), v2);
591:        }
592:    }
593:    else if (inCount == 2)
594:    {
595:        if (!in0)
596:        {
597:            const float t1{ (kNearW - v0.w) / (v1.w - v0.w) };
598:            const float t2{ (kNearW - v0.w) / (v2.w - v0.w) };
599:            const ClipVertex v01{ lerpVertex(v0, v1, t1) };
600:            const ClipVertex v02{ lerpVertex(v0, v2, t2) };
601:            projectAndDraw(v01, v1, v2);
602:            projectAndDraw(v01, v2, v02);
603:        }
604:        else if (!in1)
605:        {
606:            const float t0{ (kNearW - v1.w) / (v0.w - v1.w) };
607:            const float t2{ (kNearW - v1.w) / (v2.w - v1.w) };
608:            const ClipVertex v10{ lerpVertex(v1, v0, t0) };
609:            const ClipVertex v12{ lerpVertex(v1, v2, t2) };
610:            projectAndDraw(v0, v10, v2);
611:            projectAndDraw(v10, v12, v2);
612:        }
613:        else
614:        {
615:            const float t0{ (kNearW - v2.w) / (v0.w - v2.w) };
616:            const float t1{ (kNearW - v2.w) / (v1.w - v2.w) };
617:            const ClipVertex v20{ lerpVertex(v2, v0, t0) };
618:            const ClipVertex v21{ lerpVertex(v2, v1, t1) };
619:            projectAndDraw(v0, v1, v21);
620:            projectAndDraw(v0, v21, v20);
621:        }
622:    }

625:void ModelRenderer::SoftwareOcclusionBuffer::RasterizeOccluderMesh(
626:    const Model::Mesh& mesh,
627:    const DirectX::XMFLOAT4X4& worldMatrix,
628:    const DirectX::XMFLOAT4X4& viewProj) noexcept
629:{
630:    if (mesh.vertices.empty() || mesh.indices.empty()) return;

640:    {

643:        const auto& p2{ mesh.vertices[mesh.indices[i + 2]].position };
645:        const ClipVertex v0{
646:            p0.x * M._11 + p0.y * M._21 + p0.z * M._31 + M._41,
647:            p0.x * M._12 + p0.y * M._22 + p0.z * M._32 + M._42,
648:            p0.x * M._13 + p0.y * M._23 + p0.z * M._33 + M._43,
649:            p0.x * M._14 + p0.y * M._24 + p0.z * M._34 + M._44

652:        const ClipVertex v1{
653:            p1.x * M._11 + p1.y * M._21 + p1.z * M._31 + M._41,
654:            p1.x * M._12 + p1.y * M._22 + p1.z * M._32 + M._42,
655:            p1.x * M._13 + p1.y * M._23 + p1.z * M._33 + M._43,
656:            p1.x * M._14 + p1.y * M._24 + p1.z * M._34 + M._44

662:            p2.x * M._13 + p2.y * M._23 + p2.z * M._33 + M._43,
663:            p2.x * M._14 + p2.y * M._24 + p2.z * M._34 + M._44

670:bool ModelRenderer::SoftwareOcclusionBuffer::IsOccluded(
671:    const DirectX::XMFLOAT3& center,
672:    const DirectX::XMFLOAT3& extents,
673:    const DirectX::XMFLOAT4X4& viewProj) const noexcept
674:{
675:    const DirectX::XMFLOAT3 corners[8]{
676:        { center.x - extents.x, center.y - extents.y, center.z - extents.z },
677:        { center.x + extents.x, center.y - extents.y, center.z - extents.z },
678:        { center.x + extents.x, center.y + extents.y, center.z - extents.z },
679:        { center.x - extents.x, center.y + extents.y, center.z - extents.z },
680:        { center.x - extents.x, center.y - extents.y, center.z + extents.z },
681:        { center.x + extents.x, center.y - extents.y, center.z + extents.z },
682:        { center.x + extents.x, center.y + extents.y, center.z + extents.z },
683:        { center.x - extents.x, center.y + extents.y, center.z + extents.z }
684:    };
686:    float minX{ 1e9f }, maxX{ -1e9f };
687:    float minY{ 1e9f }, maxY{ -1e9f };
688:    float minZ{ 1e9f };
690:    for (int i{ 0 }; i < 8; ++i)
691:    {
692:        float sx{ 0.0f }, sy{ 0.0f }, sz{ 0.0f };
693:        if (!ProjectPoint(corners[i], viewProj, sx, sy, sz))
694:        {
695:            return false;
696:        }
698:        minX = (std::min)(minX, sx);
699:        maxX = (std::max)(maxX, sx);
700:        minY = (std::min)(minY, sy);
701:        maxY = (std::max)(maxY, sy);
702:        minZ = (std::min)(minZ, sz);
703:    }
705:    int startX{ std::clamp(static_cast<int>(std::floor(minX)), 0, kWidth - 1) };
706:    int endX{ std::clamp(static_cast<int>(std::ceil(maxX)), 0, kWidth - 1) };
707:    int startY{ std::clamp(static_cast<int>(std::floor(minY)), 0, kHeight - 1) };
708:    int endY{ std::clamp(static_cast<int>(std::ceil(maxY)), 0, kHeight - 1) };
710:    if (startX > endX || startY > endY) return false;
712:    // Strict conservative test: every pixel in the footprint must have occluder depth closer than the occludee's nearest depth
713:    for (int y{ startY }; y <= endY; ++y)
714:    {
715:        const int rowOffset{ y * kWidth };
716:        for (int x{ startX }; x <= endX; ++x)
717:        {
718:            if (depthBuffer[rowOffset + x] >= minZ)
719:            {
720:                return false;
721:            }
722:        }
723:    }
725:    return true;
726:}

768:void ModelRenderer::ComputeAndUploadSkeleton(
769:    ID3D11DeviceContext* dc, std::size_t slotIndex, const Model::Mesh& mesh, bool useManual,
770:    bool needsPrevious,
771:    const DirectX::XMFLOAT4X4& worldMatrix, const DirectX::XMFLOAT4X4& previousWorldMatrix,
772:    const std::vector<DirectX::XMFLOAT4X4>* currentNodeGlobals,
773:    const std::vector<DirectX::XMFLOAT4X4>* previousNodeGlobals)
774:{
775:    // Bypass buffer writes entirely if this slot is the shared global identity slot
776:    if (slotIndex == m_identitySkeletonSlot) return;
778:    SkeletonSlot& slot{ m_skeletonPool[slotIndex] };
780:    CbSkeleton cbCurrent{};
781:    CbSkeleton cbPrevious{}; // only populated and uploaded when needsPrevious is true
783:    const DirectX::XMMATRIX manualWorldMat{ DirectX::XMLoadFloat4x4(&worldMatrix) };
784:    const DirectX::XMMATRIX previousManualWorldMat{ DirectX::XMLoadFloat4x4(&previousWorldMatrix) };
786:    if (!mesh.bones.empty())
787:    {
788:        for (std::size_t i{ 0 }; i < mesh.bones.size(); ++i)
789:        {
790:            const Model::Bone& bone{ mesh.bones[i] };
792:            DirectX::XMMATRIX nodeGlobalMat;
793:            if (currentNodeGlobals && bone.nodeIndex >= 0 && static_cast<std::size_t>(bone.nodeIndex) < currentNodeGlobals->size())
794:            {
795:                nodeGlobalMat = DirectX::XMLoadFloat4x4(&(*currentNodeGlobals)[bone.nodeIndex]);
796:            }
797:            else
798:            {
799:                nodeGlobalMat = DirectX::XMLoadFloat4x4(&bone.node->globalTransform);
800:            }
802:            const DirectX::XMMATRIX offsetTransform{ DirectX::XMLoadFloat4x4(&bone.offsetTransform) };
803:            const DirectX::XMMATRIX worldTransform{ useManual
804:                ? (nodeGlobalMat * manualWorldMat)
805:                : DirectX::XMLoadFloat4x4(&bone.node->worldTransform) };
806:            DirectX::XMStoreFloat4x4(&cbCurrent.boneTransforms[i], offsetTransform * worldTransform);
808:            if (needsPrevious)
809:            {
810:                DirectX::XMMATRIX prevNodeGlobalMat;
811:                if (previousNodeGlobals && bone.nodeIndex >= 0 && static_cast<std::size_t>(bone.nodeIndex) < previousNodeGlobals->size())
812:                {
813:                    prevNodeGlobalMat = DirectX::XMLoadFloat4x4(&(*previousNodeGlobals)[bone.nodeIndex]);
814:                }
815:                else
816:                {
817:                    prevNodeGlobalMat = nodeGlobalMat;
818:                }
820:                const DirectX::XMMATRIX previousWorldTransform{ useManual
821:                    ? (prevNodeGlobalMat * previousManualWorldMat)
822:                    : DirectX::XMLoadFloat4x4(&bone.node->worldTransform) };
823:                DirectX::XMStoreFloat4x4(&cbPrevious.boneTransforms[i], offsetTransform * previousWorldTransform);
824:            }
825:        }
826:    }
827:    else
828:    {
829:        if (useManual)
830:        {
831:            // Unskinned meshes must track the animated node
832:            DirectX::XMMATRIX nodeGlobalMat;
834:            // If the AnimationComponent provided evaluated globals, use them
835:            if (currentNodeGlobals && mesh.nodeIndex >= 0 && static_cast<std::size_t>(mesh.nodeIndex) < currentNodeGlobals->size())
836:            {
837:                nodeGlobalMat = DirectX::XMLoadFloat4x4(&(*currentNodeGlobals)[mesh.nodeIndex]);
838:            }
839:            else
840:            {
841:                // Fallback to static bind pose if no AnimationComponent exists
842:                nodeGlobalMat = DirectX::XMLoadFloat4x4(&mesh.node->globalTransform);
843:            }
845:            DirectX::XMStoreFloat4x4(&cbCurrent.boneTransforms[0], nodeGlobalMat * manualWorldMat);
847:            if (needsPrevious)
848:            {
849:                DirectX::XMMATRIX prevNodeGlobalMat;
850:                if (previousNodeGlobals && mesh.nodeIndex >= 0 && static_cast<std::size_t>(mesh.nodeIndex) < previousNodeGlobals->size())
851:                {
852:                    prevNodeGlobalMat = DirectX::XMLoadFloat4x4(&(*previousNodeGlobals)[mesh.nodeIndex]);
853:                }
854:                else
855:                {
856:                    prevNodeGlobalMat = nodeGlobalMat;
857:                }
859:                DirectX::XMStoreFloat4x4(&cbPrevious.boneTransforms[0], prevNodeGlobalMat * previousManualWorldMat);
860:            }
861:        }
862:        else
863:        {
864:            cbCurrent.boneTransforms[0] = mesh.node->worldTransform;
865:            if (needsPrevious)
866:            {
867:                cbPrevious.boneTransforms[0] = mesh.node->worldTransform;
868:            }
869:        }
870:    }
872:    dc->UpdateSubresource(slot.currentBuffer.Get(), 0, nullptr, &cbCurrent, 0, 0);
873:    if (needsPrevious)
874:    {
875:        dc->UpdateSubresource(slot.previousBuffer.Get(), 0, nullptr, &cbPrevious, 0, 0);
876:    }

939:{
940:    if (drawInfos.empty()) return;

953:        SkeletonSlot& idSlot = m_skeletonPool[m_identitySkeletonSlot];
954:        CbSkeleton cbId{};

956:        for (int i = 0; i < 256; ++i)
957:        {

972:        {
973:            if (meshes[meshIdx].bones.empty())
974:            {

976:            }
977:            else
978:            {

980:                // If an earlier mesh in this drawInfo has an identical bone palette, reuse its uploaded slot
981:                std::size_t sharedSlot{ kInvalidSkeletonSlot };
982:                for (std::size_t prevIdx{ 0 }; prevIdx < meshIdx; ++prevIdx)

989:                        break;
990:                    }
