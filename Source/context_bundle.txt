260:        struct alignas(16) MaterialKeyData

284:        switch (static_cast<ShaderId>(mat->shaderId))

295:            data.normalMap = mat->normalMap.Get();
296:            data.emissiveColor = mat->emissiveColor;
297:            data.roughness = mat->roughness;
298:            break;
300:        case ShaderId::Pbr:
301:            data.normalMap = mat->normalMap.Get();
302:            data.emissiveMap = mat->emissiveMap.Get();
303:            data.occlusionMap = mat->occlusionMap.Get();
304:            data.metalnessRoughnessMap = mat->metalnessRoughnessMap.Get();

356:    [[nodiscard]] bool AreMaterialsEquivalent(const Material* a, const Material* b, ShaderId shaderId) noexcept
357:    {
358:        if (a == b) return true;
359:        if (!a || !b) return false;
360:        if (a->baseMap.Get() != b->baseMap.Get()) return false;
361:        if (a->alphaMode != b->alphaMode || a->alphaCutoff != b->alphaCutoff) return false;
363:        switch (shaderId)
364:        {
365:        case ShaderId::Basic:
366:            return true;
368:        case ShaderId::Lambert:
369:            return (a->emissiveColor.x == b->emissiveColor.x &&
370:                a->emissiveColor.y == b->emissiveColor.y &&
371:                a->emissiveColor.z == b->emissiveColor.z);
373:        case ShaderId::Phong:
374:            return (a->normalMap.Get() == b->normalMap.Get() &&
375:                a->roughness == b->roughness &&
376:                a->emissiveColor.x == b->emissiveColor.x &&
377:                a->emissiveColor.y == b->emissiveColor.y &&
378:                a->emissiveColor.z == b->emissiveColor.z);
380:        case ShaderId::Pbr:
381:            return (a->normalMap.Get() == b->normalMap.Get() &&
382:                a->metalnessRoughnessMap.Get() == b->metalnessRoughnessMap.Get() &&
383:                a->emissiveMap.Get() == b->emissiveMap.Get() &&
384:                a->occlusionMap.Get() == b->occlusionMap.Get() &&
385:                a->roughness == b->roughness &&
386:                a->metalness == b->metalness &&
387:                a->occlusionStrength == b->occlusionStrength &&
388:                a->emissiveColor.x == b->emissiveColor.x &&
389:                a->emissiveColor.y == b->emissiveColor.y &&
390:                a->emissiveColor.z == b->emissiveColor.z);
392:        case ShaderId::Toon:
393:            return (a->roughness == b->roughness &&
394:                a->emissiveColor.x == b->emissiveColor.x &&
395:                a->emissiveColor.y == b->emissiveColor.y &&
396:                a->emissiveColor.z == b->emissiveColor.z);
398:        default:
399:            return false;
400:        }
401:    }
403:    // Evaluates semantic property equality strictly for outline parameters
404:    [[nodiscard]] bool AreOutlineMaterialsEquivalent(const Material* a, const Material* b) noexcept
405:    {
406:        if (a == b) return true;
407:        if (!a || !b) return false;
408:        return (a->baseMap.Get() == b->baseMap.Get() &&
409:            a->outlineColor.x == b->outlineColor.x &&
410:            a->outlineColor.y == b->outlineColor.y &&
411:            a->outlineColor.z == b->outlineColor.z &&
412:            a->outlineColor.w == b->outlineColor.w &&
413:            a->outlineWidth == b->outlineWidth &&
414:            a->outlineFadeStart == b->outlineFadeStart &&
415:            a->outlineFadeEnd == b->outlineFadeEnd &&
416:            a->alphaMode == b->alphaMode &&
417:            a->alphaCutoff == b->alphaCutoff);
418:    }

426:bool ModelRenderer::SoftwareOcclusionBuffer::ProjectPoint(
427:    const DirectX::XMFLOAT3& worldPos,
428:    const DirectX::XMFLOAT4X4& viewProj,
429:    float& outX, float& outY, float& outZ) noexcept
430:{
431:    const float w{ worldPos.x * viewProj._14 + worldPos.y * viewProj._24 + worldPos.z * viewProj._34 + viewProj._44 };
432:    if (w <= 0.001f) return false;
434:    const float invW{ 1.0f / w };
435:    const float ndcX{ (worldPos.x * viewProj._11 + worldPos.y * viewProj._21 + worldPos.z * viewProj._31 + viewProj._41) * invW };
436:    const float ndcY{ (worldPos.x * viewProj._12 + worldPos.y * viewProj._22 + worldPos.z * viewProj._32 + viewProj._42) * invW };
437:    const float ndcZ{ (worldPos.x * viewProj._13 + worldPos.y * viewProj._23 + worldPos.z * viewProj._33 + viewProj._43) * invW };
439:    outX = (ndcX * 0.5f + 0.5f) * static_cast<float>(kWidth - 1);
440:    outY = (-ndcY * 0.5f + 0.5f) * static_cast<float>(kHeight - 1);
441:    outZ = ndcZ;
442:    return true;

445:void ModelRenderer::SoftwareOcclusionBuffer::RasterizeTriangle(
446:    float x0, float y0, float z0,
447:    float x1, float y1, float z1,
448:    float x2, float y2, float z2) noexcept
449:{
450:    float area{ (x1 - x0) * (y2 - y0) - (y1 - y0) * (x2 - x0) };
451:    if (std::abs(area) < 0.0001f)
452:    {
453:        return;
454:    }
456:    // Two-sided support for walls: ensure consistent clockwise rasterization winding
457:    if (area < 0.0f)
458:    {
459:        std::swap(x1, x2);
460:        std::swap(y1, y2);
461:        std::swap(z1, z2);
462:        area = -area;
463:    }
465:    int minX{ static_cast<int>(std::floor((std::min)({ x0, x1, x2 }))) };
466:    int maxX{ static_cast<int>(std::ceil((std::max)({ x0, x1, x2 }))) };
467:    int minY{ static_cast<int>(std::floor((std::min)({ y0, y1, y2 }))) };
468:    int maxY{ static_cast<int>(std::ceil((std::max)({ y0, y1, y2 }))) };
470:    minX = std::clamp(minX, 0, kWidth - 1);
471:    maxX = std::clamp(maxX, 0, kWidth - 1);
472:    minY = std::clamp(minY, 0, kHeight - 1);
473:    maxY = std::clamp(maxY, 0, kHeight - 1);
475:    if (minX > maxX || minY > maxY) return;
477:    const float invArea{ 1.0f / area };
479:    for (int y{ minY }; y <= maxY; ++y)
480:    {
481:        const float py{ static_cast<float>(y) + 0.5f };
482:        const int rowOffset{ y * kWidth };
484:        for (int x{ minX }; x <= maxX; ++x)
485:        {
486:            const float px{ static_cast<float>(x) + 0.5f };
487:            const float w0{ ((x1 - px) * (y2 - py) - (y1 - py) * (x2 - px)) * invArea };
488:            const float w1{ ((x2 - px) * (y0 - py) - (y2 - py) * (x0 - px)) * invArea };
489:            const float w2{ 1.0f - w0 - w1 };
491:            if (w0 >= 0.0f && w1 >= 0.0f && w2 >= 0.0f)
492:            {
493:                const float depth{ w0 * z0 + w1 * z1 + w2 * z2 };
494:                const int idx{ rowOffset + x };
495:                if (depth < depthBuffer[idx])
496:                {
497:                    depthBuffer[idx] = depth;
498:                }
499:            }
500:        }
501:    }
502:}
504:void ModelRenderer::SoftwareOcclusionBuffer::ClipAndRasterizeTriangle(
505:    const ClipVertex& v0,
506:    const ClipVertex& v1,
507:    const ClipVertex& v2) noexcept
508:{
509:    constexpr float kNearW{ 0.05f };
511:    auto projectAndDraw = [this](const ClipVertex& a, const ClipVertex& b, const ClipVertex& c) noexcept
512:        {
513:            const float invWa{ 1.0f / a.w };
514:            const float invWb{ 1.0f / b.w };
515:            const float invWc{ 1.0f / c.w };
517:            const float sx0{ (a.x * invWa * 0.5f + 0.5f) * static_cast<float>(kWidth - 1) };
518:            const float sy0{ (-a.y * invWa * 0.5f + 0.5f) * static_cast<float>(kHeight - 1) };
519:            const float sz0{ a.z * invWa };
521:            const float sx1{ (b.x * invWb * 0.5f + 0.5f) * static_cast<float>(kWidth - 1) };
522:            const float sy1{ (-b.y * invWb * 0.5f + 0.5f) * static_cast<float>(kHeight - 1) };
523:            const float sz1{ b.z * invWb };
525:            const float sx2{ (c.x * invWc * 0.5f + 0.5f) * static_cast<float>(kWidth - 1) };
526:            const float sy2{ (-c.y * invWc * 0.5f + 0.5f) * static_cast<float>(kHeight - 1) };
527:            const float sz2{ c.z * invWc };
529:            RasterizeTriangle(sx0, sy0, sz0, sx1, sy1, sz1, sx2, sy2, sz2);
530:        };
532:    auto lerpVertex = [](const ClipVertex& a, const ClipVertex& b, float t) noexcept -> ClipVertex
533:        {
534:            return {
535:                a.x + t * (b.x - a.x),
536:                a.y + t * (b.y - a.y),
537:                a.z + t * (b.z - a.z),
538:                a.w + t * (b.w - a.w)
539:            };
540:        };
542:    const bool in0{ v0.w >= kNearW };
543:    const bool in1{ v1.w >= kNearW };
544:    const bool in2{ v2.w >= kNearW };
545:    const int inCount{ (in0 ? 1 : 0) + (in1 ? 1 : 0) + (in2 ? 1 : 0) };
547:    if (inCount == 3)
548:    {
549:        projectAndDraw(v0, v1, v2);
550:    }
551:    else if (inCount == 1)
552:    {
553:        if (in0)
554:        {
555:            const float t1{ (kNearW - v0.w) / (v1.w - v0.w) };
556:            const float t2{ (kNearW - v0.w) / (v2.w - v0.w) };
557:            projectAndDraw(v0, lerpVertex(v0, v1, t1), lerpVertex(v0, v2, t2));
558:        }
559:        else if (in1)
560:        {
561:            const float t0{ (kNearW - v1.w) / (v0.w - v1.w) };
562:            const float t2{ (kNearW - v1.w) / (v2.w - v1.w) };
563:            projectAndDraw(lerpVertex(v1, v0, t0), v1, lerpVertex(v1, v2, t2));
564:        }
565:        else
566:        {
567:            const float t0{ (kNearW - v2.w) / (v0.w - v2.w) };
568:            const float t1{ (kNearW - v2.w) / (v1.w - v2.w) };
569:            projectAndDraw(lerpVertex(v2, v0, t0), lerpVertex(v2, v1, t1), v2);
570:        }
571:    }
572:    else if (inCount == 2)
573:    {
574:        if (!in0)
575:        {
576:            const float t1{ (kNearW - v0.w) / (v1.w - v0.w) };
577:            const float t2{ (kNearW - v0.w) / (v2.w - v0.w) };
578:            const ClipVertex v01{ lerpVertex(v0, v1, t1) };
579:            const ClipVertex v02{ lerpVertex(v0, v2, t2) };
580:            projectAndDraw(v01, v1, v2);
581:            projectAndDraw(v01, v2, v02);
582:        }
583:        else if (!in1)
584:        {
585:            const float t0{ (kNearW - v1.w) / (v0.w - v1.w) };
586:            const float t2{ (kNearW - v1.w) / (v2.w - v1.w) };
587:            const ClipVertex v10{ lerpVertex(v1, v0, t0) };
588:            const ClipVertex v12{ lerpVertex(v1, v2, t2) };
589:            projectAndDraw(v0, v10, v2);
590:            projectAndDraw(v10, v12, v2);
591:        }
592:        else
593:        {
594:            const float t0{ (kNearW - v2.w) / (v0.w - v2.w) };
595:            const float t1{ (kNearW - v2.w) / (v1.w - v2.w) };
596:            const ClipVertex v20{ lerpVertex(v2, v0, t0) };
597:            const ClipVertex v21{ lerpVertex(v2, v1, t1) };
598:            projectAndDraw(v0, v1, v21);
599:            projectAndDraw(v0, v21, v20);
600:        }
601:    }

604:void ModelRenderer::SoftwareOcclusionBuffer::RasterizeOccluderMesh(
605:    const Model::Mesh& mesh,
606:    const DirectX::XMFLOAT4X4& worldMatrix,
607:    const DirectX::XMFLOAT4X4& viewProj) noexcept
608:{
609:    if (mesh.vertices.empty() || mesh.indices.empty()) return;
611:    const DirectX::XMMATRIX W{ DirectX::XMLoadFloat4x4(&worldMatrix) };
612:    const DirectX::XMMATRIX VP{ DirectX::XMLoadFloat4x4(&viewProj) };
613:    const DirectX::XMMATRIX WVP{ W * VP };
614:    DirectX::XMFLOAT4X4 M{};
615:    DirectX::XMStoreFloat4x4(&M, WVP);
617:    const std::size_t indexCount{ mesh.indices.size() };
618:    for (std::size_t i{ 0 }; i + 2 < indexCount; i += 3)
619:    {
620:        const auto& p0{ mesh.vertices[mesh.indices[i]].position };
621:        const auto& p1{ mesh.vertices[mesh.indices[i + 1]].position };
622:        const auto& p2{ mesh.vertices[mesh.indices[i + 2]].position };
624:        const ClipVertex v0{
625:            p0.x * M._11 + p0.y * M._21 + p0.z * M._31 + M._41,
626:            p0.x * M._12 + p0.y * M._22 + p0.z * M._32 + M._42,
627:            p0.x * M._13 + p0.y * M._23 + p0.z * M._33 + M._43,
628:            p0.x * M._14 + p0.y * M._24 + p0.z * M._34 + M._44
629:        };
631:        const ClipVertex v1{
632:            p1.x * M._11 + p1.y * M._21 + p1.z * M._31 + M._41,
633:            p1.x * M._12 + p1.y * M._22 + p1.z * M._32 + M._42,
634:            p1.x * M._13 + p1.y * M._23 + p1.z * M._33 + M._43,
635:            p1.x * M._14 + p1.y * M._24 + p1.z * M._34 + M._44
636:        };
638:        const ClipVertex v2{
639:            p2.x * M._11 + p2.y * M._21 + p2.z * M._31 + M._41,
640:            p2.x * M._12 + p2.y * M._22 + p2.z * M._32 + M._42,
641:            p2.x * M._13 + p2.y * M._23 + p2.z * M._33 + M._43,
642:            p2.x * M._14 + p2.y * M._24 + p2.z * M._34 + M._44
643:        };
645:        ClipAndRasterizeTriangle(v0, v1, v2);
646:    }

649:bool ModelRenderer::SoftwareOcclusionBuffer::IsOccluded(
650:    const DirectX::XMFLOAT3& center,
651:    const DirectX::XMFLOAT3& extents,
652:    const DirectX::XMFLOAT4X4& viewProj) const noexcept
653:{
654:    const DirectX::XMFLOAT3 corners[8]{
655:        { center.x - extents.x, center.y - extents.y, center.z - extents.z },
656:        { center.x + extents.x, center.y - extents.y, center.z - extents.z },
657:        { center.x + extents.x, center.y + extents.y, center.z - extents.z },
658:        { center.x - extents.x, center.y + extents.y, center.z - extents.z },
659:        { center.x - extents.x, center.y - extents.y, center.z + extents.z },
660:        { center.x + extents.x, center.y - extents.y, center.z + extents.z },
661:        { center.x + extents.x, center.y + extents.y, center.z + extents.z },
662:        { center.x - extents.x, center.y + extents.y, center.z + extents.z }
663:    };
665:    float minX{ 1e9f }, maxX{ -1e9f };
666:    float minY{ 1e9f }, maxY{ -1e9f };
667:    float minZ{ 1e9f };
669:    for (int i{ 0 }; i < 8; ++i)
670:    {
671:        float sx{ 0.0f }, sy{ 0.0f }, sz{ 0.0f };
672:        if (!ProjectPoint(corners[i], viewProj, sx, sy, sz))
673:        {
674:            return false;
675:        }
677:        minX = (std::min)(minX, sx);
678:        maxX = (std::max)(maxX, sx);
679:        minY = (std::min)(minY, sy);
680:        maxY = (std::max)(maxY, sy);
681:        minZ = (std::min)(minZ, sz);
682:    }
684:    int startX{ std::clamp(static_cast<int>(std::floor(minX)), 0, kWidth - 1) };
685:    int endX{ std::clamp(static_cast<int>(std::ceil(maxX)), 0, kWidth - 1) };
686:    int startY{ std::clamp(static_cast<int>(std::floor(minY)), 0, kHeight - 1) };
687:    int endY{ std::clamp(static_cast<int>(std::ceil(maxY)), 0, kHeight - 1) };
689:    if (startX > endX || startY > endY) return false;
691:    // Strict conservative test: every pixel in the footprint must have occluder depth closer than the occludee's nearest depth
692:    for (int y{ startY }; y <= endY; ++y)
693:    {
694:        const int rowOffset{ y * kWidth };
695:        for (int x{ startX }; x <= endX; ++x)
696:        {
697:            if (depthBuffer[rowOffset + x] >= minZ)
698:            {
699:                return false;
700:            }
701:        }
702:    }
704:    return true;
705:}

747:void ModelRenderer::ComputeAndUploadSkeleton(
748:    ID3D11DeviceContext* dc, std::size_t slotIndex, const Model::Mesh& mesh, bool useManual,
749:    bool needsPrevious,
750:    const DirectX::XMFLOAT4X4& worldMatrix, const DirectX::XMFLOAT4X4& previousWorldMatrix,
751:    const std::vector<DirectX::XMFLOAT4X4>* currentNodeGlobals,
752:    const std::vector<DirectX::XMFLOAT4X4>* previousNodeGlobals)
753:{
754:    // Bypass buffer writes entirely if this slot is the shared global identity slot
755:    if (slotIndex == m_identitySkeletonSlot) return;
757:    SkeletonSlot& slot{ m_skeletonPool[slotIndex] };
759:    CbSkeleton cbCurrent{};
760:    CbSkeleton cbPrevious{}; // only populated and uploaded when needsPrevious is true
762:    const DirectX::XMMATRIX manualWorldMat{ DirectX::XMLoadFloat4x4(&worldMatrix) };
763:    const DirectX::XMMATRIX previousManualWorldMat{ DirectX::XMLoadFloat4x4(&previousWorldMatrix) };
765:    if (!mesh.bones.empty())
766:    {
767:        for (std::size_t i{ 0 }; i < mesh.bones.size(); ++i)
768:        {
769:            const Model::Bone& bone{ mesh.bones[i] };
771:            DirectX::XMMATRIX nodeGlobalMat;
772:            if (currentNodeGlobals && bone.nodeIndex >= 0 && static_cast<std::size_t>(bone.nodeIndex) < currentNodeGlobals->size())
773:            {
774:                nodeGlobalMat = DirectX::XMLoadFloat4x4(&(*currentNodeGlobals)[bone.nodeIndex]);
775:            }
776:            else
777:            {
778:                nodeGlobalMat = DirectX::XMLoadFloat4x4(&bone.node->globalTransform);
779:            }
781:            const DirectX::XMMATRIX offsetTransform{ DirectX::XMLoadFloat4x4(&bone.offsetTransform) };
782:            const DirectX::XMMATRIX worldTransform{ useManual
783:                ? (nodeGlobalMat * manualWorldMat)
784:                : DirectX::XMLoadFloat4x4(&bone.node->worldTransform) };
785:            DirectX::XMStoreFloat4x4(&cbCurrent.boneTransforms[i], offsetTransform * worldTransform);
787:            if (needsPrevious)
788:            {
789:                DirectX::XMMATRIX prevNodeGlobalMat;
790:                if (previousNodeGlobals && bone.nodeIndex >= 0 && static_cast<std::size_t>(bone.nodeIndex) < previousNodeGlobals->size())
791:                {
792:                    prevNodeGlobalMat = DirectX::XMLoadFloat4x4(&(*previousNodeGlobals)[bone.nodeIndex]);
793:                }
794:                else
795:                {
796:                    prevNodeGlobalMat = nodeGlobalMat;
797:                }
799:                const DirectX::XMMATRIX previousWorldTransform{ useManual
800:                    ? (prevNodeGlobalMat * previousManualWorldMat)
801:                    : DirectX::XMLoadFloat4x4(&bone.node->worldTransform) };
802:                DirectX::XMStoreFloat4x4(&cbPrevious.boneTransforms[i], offsetTransform * previousWorldTransform);
803:            }
804:        }
805:    }
806:    else
807:    {
808:        if (useManual)
809:        {
810:            // Unskinned meshes must track the animated node
811:            DirectX::XMMATRIX nodeGlobalMat;
813:            // If the AnimationComponent provided evaluated globals, use them
814:            if (currentNodeGlobals && mesh.nodeIndex >= 0 && static_cast<std::size_t>(mesh.nodeIndex) < currentNodeGlobals->size())
815:            {
816:                nodeGlobalMat = DirectX::XMLoadFloat4x4(&(*currentNodeGlobals)[mesh.nodeIndex]);
817:            }
818:            else
819:            {
820:                // Fallback to static bind pose if no AnimationComponent exists
821:                nodeGlobalMat = DirectX::XMLoadFloat4x4(&mesh.node->globalTransform);
822:            }
824:            DirectX::XMStoreFloat4x4(&cbCurrent.boneTransforms[0], nodeGlobalMat * manualWorldMat);
826:            if (needsPrevious)
827:            {
828:                DirectX::XMMATRIX prevNodeGlobalMat;
829:                if (previousNodeGlobals && mesh.nodeIndex >= 0 && static_cast<std::size_t>(mesh.nodeIndex) < previousNodeGlobals->size())
830:                {
831:                    prevNodeGlobalMat = DirectX::XMLoadFloat4x4(&(*previousNodeGlobals)[mesh.nodeIndex]);
832:                }
833:                else
834:                {
835:                    prevNodeGlobalMat = nodeGlobalMat;
836:                }
838:                DirectX::XMStoreFloat4x4(&cbPrevious.boneTransforms[0], prevNodeGlobalMat * previousManualWorldMat);
839:            }
840:        }
841:        else
842:        {
843:            cbCurrent.boneTransforms[0] = mesh.node->worldTransform;
844:            if (needsPrevious)
845:            {
846:                cbPrevious.boneTransforms[0] = mesh.node->worldTransform;
847:            }
